event ObjectDraw
	rain.scroll++
	rain.scroll &= 63
	temp0 = rain.scroll
	while temp0 < screen.ysize
		temp1 = 0
		while temp1 < screen.xsize
			DrawSpriteScreenFX(0, FX_INK, temp1, temp0)
			temp1 += 64		
		loop
		temp0 += 64
	loop	

	object.InkEffect = INK_ALPHA
	object.alpha = 96

	DrawRect(0, 0, screen.xsize, screen.ysize, 0x00, 0x00, 0x00, 255)
end event
